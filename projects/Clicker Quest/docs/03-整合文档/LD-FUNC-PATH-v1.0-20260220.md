#### 2.1.2 点击数据计算

```
输入参数:
- baseGoldPerClick: 基础每次点击金币 = 1
- clickMultiplier: 点击倍率加成 = 1 + 道具加成
- critChance: 暴击率 = 5% + 道具加成
- critMultiplier: 暴击倍率 = 2x/5x/10x

计算公式:
goldEarned = baseGoldPerClick × clickMultiplier × critMultiplier

暴击判定:
random = Random(0, 100)
if random < critChance:
    if random < 1%: critMultiplier = 10x  // 传说暴击
    elif random < 5%: critMultiplier = 5x  // 稀有暴击
    else: critMultiplier = 2x              // 普通暴击
else:
    critMultiplier = 1x
```

### 4.3 道具叠加规则

```
叠加规则判定:
┌──────────────────────────────────────────────┐
│              道具叠加类型                     │
├────────────┬────────────┬────────────┬───────┤
│ STACKABLE  │ REFRESH    │ CAP        │ NON_  │
│ (可叠加)   │ (刷新时长) │ (上限叠加) │ STACK-│
│            │            │            │ ABLE  │
├────────────┼────────────┼────────────┼───────┤
│ 时长相加   │ 重置为     │ 叠加但     │ 独立  │
│ T = T1+T2  │ 新时长     │ 有上限     │ 存在  │
│            │ T = T_new  │ T ≤ T_max  │       │
└────────────┴────────────┴────────────┴───────┘

示例:
- 双倍金币(STACKABLE): 购买2次30分钟 = 60分钟
- 暴击提升(REFRESH): 购买2次 = 重置为30分钟
- DPS加成(CAP): 上限100%, 购买超过则截断
- 特殊道具(NON_STACKABLE): 每个独立存在
```

### C.1 点击测试脚本 (C#)

```csharp
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;
using System.Collections;

[TestFixture]
public class ClickSystemTests
{
    private ClickSystem clickSystem;
    private PlayerData playerData;

    [SetUp]
    public void Setup()
    {
        clickSystem = new ClickSystem();
        playerData = new PlayerData();
    }

    [Test]
    public void TestBasicClick()
    {
        // Arrange
        decimal initialGold = playerData.gold;
        
        // Act
        clickSystem.ProcessClick(playerData, Vector2.zero);
        
        // Assert
        Assert.AreEqual(initialGold + 1, playerData.gold);
    }

    [Test]
    public void TestCriticalHit()
    {
        // Arrange
        int critCount = 0;
        int totalClicks = 1000;
        
        // Act
        for (int i = 0; i < totalClicks; i++)
        {
            if (clickSystem.CheckCritical(5)) // 5% crit chance
            {
                critCount++;
            }
        }
        
        // Assert - 暴击率应在3%-7%范围内(允许误差)
        float critRate = (float)critCount / totalClicks * 100;
        Assert.IsTrue(critRate >= 3 && critRate <= 7, 
            $"Crit rate {critRate}% is out of expected range [3%, 7%]");
    }

    [UnityTest]
    public IEnumerator TestAutoProduction()
    {
        // Arrange
        playerData.dps = 100;
        decimal initialGold = playerData.gold;
        
        // Act
        yield return new WaitForSeconds(1f);
        clickSystem.ProcessAutoProduction(playerData);
        
        // Assert
        Assert.AreEqual(initialGold + 100, playerData.gold);
    }
}
```

### C.2 商店测试脚本 (TypeScript)

```typescript
describe('ShopSystem', () => {
    let shopSystem: ShopSystem;
    let playerData: PlayerData;

    beforeEach(() => {
        shopSystem = new ShopSystem();
        playerData = {
            gold: 10000,
            dps: 0,
            items: [],
            achievements: []
        };
    });

    test('should purchase item successfully', () => {
        // Arrange
        const item = {
            id: 'AUTO_001',
            name: '小助手',
            type: 'AUTO',
            price: 1000,
            dpsBonus: 10
        };

        // Act
        const result = shopSystem.purchaseItem(playerData, item);

        // Assert
        expect(result.success).toBe(true);
        expect(playerData.gold).toBe(9000);
        expect(playerData.dps).toBe(10);
    });

    test('should fail when gold is insufficient', () => {
        // Arrange
        playerData.gold = 500;
        const item = {
            id: 'AUTO_001',
            name: '小助手',
            type: 'AUTO',
            price: 1000,
            dpsBonus: 10
        };

        // Act
        const result = shopSystem.purchaseItem(playerData, item);

        // Assert
        expect(result.success).toBe(false);
        expect(result.error).toBe('INSUFFICIENT_GOLD');
        expect(playerData.gold).toBe(500); // 金币未扣除
    });

    test('should enforce purchase limit', () => {
        // Arrange
        const item = {
            id: 'LIMITED_001',
            name: '限量商品',
            type: 'AUTO',
            price: 100,
            dpsBonus: 5,
            maxPurchases: 3
        };

        // Act & Assert
        for (let i = 0; i < 3; i++) {
            const result = shopSystem.purchaseItem(playerData, item);
            expect(result.success).toBe(true);
        }

        const fourthPurchase = shopSystem.purchaseItem(playerData, item);
        expect(fourthPurchase.success).toBe(false);
        expect(fourthPurchase.error).toBe('PURCHASE_LIMIT_REACHED');
    });
});
```
